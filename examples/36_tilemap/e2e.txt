# E2E: tilemap draw + solid resolve (deterministic CRC baked in)
# Preconditions: 640×360 / fixed map / FixedTimeStep 60Hz / harness virtual clock
# CRC oracle is the aarch64-darwin capture. Re-capture on other platforms if it diverges.
# Example run:
#   TMPD=$(mktemp -d)
#   KNGN_HEADLESS=1 KNGN_HARNESS_SCRIPT=examples/36_tilemap/e2e.txt \
#     KNGN_HARNESS_OUT=$TMPD zig build run-example_36
# snapshot with path omitted → $KNGN_HARNESS_OUT/frame_<n>.png (do not leave PNGs in the repo)

# --- 1. Initial state (falling from air) ---
step 1
snapshot fb
digest fb
expect fb crc=190FDDC2

# --- 2. Move sideways with RIGHT ---
inject key_down RIGHT
step 40
inject key_up RIGHT
snapshot fb
digest fb
expect fb crc=038E6ED4

# --- 3. Release key: gravity fall and land ---
step 50
snapshot fb
digest fb
expect fb crc=DACCE791

# --- 4. Move the other way with LEFT ---
inject key_down LEFT
step 50
inject key_up LEFT
snapshot fb
digest fb
expect fb crc=0B01C9D9

# --- 5. Move toward the ledge and settle on land ---
inject key_down RIGHT
step 80
inject key_up RIGHT
step 20
snapshot fb
digest fb
expect fb crc=2E224B4B
quit
