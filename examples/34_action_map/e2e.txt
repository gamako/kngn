# TASK-111.8 E2E: ActionMap keyboard + gamepad + rebind
# float 値は probe 実装後に実測焼き込み（手計算と一致確認済み）。
# 実行例:
#   TMPD=$(mktemp -d)
#   VP_HEADLESS=1 VP_HARNESS_SCRIPT=examples/34_action_map/e2e.txt \
#     VP_HARNESS_OUT=$TMPD zig build run-example_34
# snapshot は path 省略 → $VP_HARNESS_OUT/frame_<n>.png

step 1
snapshot fb
digest action_map
assert action_map binding=wasd+pad
expect action_map move_x=0.0000

inject gamepad_connect 0 TestPad
step 1
digest gamepad
expect gamepad p0_lx=0.0000

inject key_down SPACE
step 1
digest action_map
assert action_map jump_down=1
assert action_map jump_pressed=1
step 1
digest action_map
expect action_map jump_pressed=0
inject key_up SPACE
step 1
expect action_map jump_down=0

inject gamepad_button 0 a 1
step 1
digest action_map
assert action_map jump_down=1
assert action_map jump_pressed=1
step 1
expect action_map jump_pressed=0
inject gamepad_button 0 a 0
step 1

inject key_down D
inject gamepad_axis 0 left_x 0.5
step 1
digest gamepad
expect gamepad p0_lx=0.5000
digest action_map
assert action_map move_source=stick
assert action_map move_x=0.4118

inject gamepad_axis 0 left_x 0
step 1
digest action_map
assert action_map move_source=key
assert action_map move_x=1.0000

inject key_up D
step 1
inject key_down R
step 1
inject key_up R
step 1
digest action_map
assert action_map binding=arrows+pad

inject key_down RIGHT
step 1
digest action_map
assert action_map move_source=key
assert action_map move_x=1.0000
inject key_up RIGHT
step 1

inject key_down D
step 1
assert action_map move_x=0.0000
inject key_up D
step 1

inject gamepad_axis 0 left_x -1
step 1
snapshot fb
digest action_map
assert action_map move_source=stick
assert action_map move_x=-1.0000

quit
